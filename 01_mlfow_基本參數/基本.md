```python
with mlflow.start_run() as run:   #跑模型外面要
	mlflow.tensorflow.autolog() #自動存參數
	run_id = mlflow.active_run().info.run_id #存模型的run_id

mlflow.set_experiment("power-forecasting")  #設置實驗號 ##必要
model_name = "power-forecasting-model" #模型命名
model_uri = f"runs:/{run_id}/model"	#用跑完的run_id抓url
model_details = mlflow.register_model(model_uri, model_name) #註冊模型#產生models/v1
```
